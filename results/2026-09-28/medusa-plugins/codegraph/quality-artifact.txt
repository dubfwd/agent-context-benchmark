8	
9	/**
10	 * Reads the plugin's options from the resolved config module.
11	 *
12	 * Works regardless of how the plugin is registered in `medusa-config.ts`:
13	 * the object form (`{ resolve, options }`) and the factory form
14	 * (`PosPlugin({...})`) both end up on `configModule.plugins` with the same
15	 * `options`, so we look ours up by package name.
16	 */
17	export function getPluginOptions(scope: Scope): PosPluginOptions {
18	  const configModule = scope.resolve(ContainerRegistrationKeys.CONFIG_MODULE)
19	
20	  const entry = (configModule?.plugins ?? []).find(
21	    (p: string | { resolve: string }) =>
22	      (typeof p === "string" ? p : p.resolve) === PLUGIN_NAME
23	  )
24	
25	  return (typeof entry === "object" ? (entry.options as PosPluginOptions) : undefined) ?? {}
26	}

2	// Rewrites @/ path aliases to relative paths in the compiled output.
3	// Medusa's build (swc) leaves @/ aliases unresolved; this script fixes them
4	// so the plugin works when installed without tsconfig-paths configured.
5	const fs = require("fs");
6	const path = require("path");
7	
8	// Medusa resolves plugin routes from .medusa/server/src (MEDUSA_PLUGIN_SOURCE_PATH).
9	// rootDir "." in tsconfig preserves the src/ prefix so the output lands there.
10	// @/* aliases map to src/*, which in the compiled output live at .medusa/server/src/*.
11	const outDir = path.resolve(__dirname, "../.medusa/server/src");
12	
13	function processDir(dir) {
14	  for (const entry of fs.readdirSync(dir)) {
15	    const full = path.join(dir, entry);
16	    if (fs.statSync(full).isDirectory()) {
17	      processDir(full);
18	    } else if (full.endsWith(".js") || full.endsWith(".mjs") || full.endsWith(".cjs")) {
19	      processFile(full);
20	    }
21	  }
22	}
23	
24	function processFile(file) {
25	  const src = fs.readFileSync(file, "utf8");
26	  const next = src.replace(
27	    /(['"])@\/([^'"]+)\1/g,
28	    (_, quote, alias) => {
29	      const target = path.join(outDir, alias);
30	      let rel = path.relative(path.dirname(file), target);
31	      if (!rel.startsWith(".")) rel = "./" + rel;
32	      return `${quote}${rel}${quote}`;
33	    }
34	  );
35	  if (next !== src) fs.writeFileSync(file, next);
36	}
37	
38	processDir(outDir);
39	console.log("fix-aliases: @/ imports resolved in .medusa/server");
