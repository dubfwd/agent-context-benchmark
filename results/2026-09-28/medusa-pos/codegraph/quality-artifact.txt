71	/**
72	 * Payments on an order that still have money left to give back.
73	 * Medusa caps a refund at (sum of captures - sum of refunds), so an uncaptured
74	 * payment is never refundable no matter what the order total says.
75	 */
76	export function getRefundablePayments(order: AdminOrder): RefundablePayment[] {
77	  const refundable: RefundablePayment[] = [];
78	
79	  for (const collection of order.payment_collections ?? []) {
80	    for (const payment of collection.payments ?? []) {
81	      if (!payment.id) continue;
82	
83	      const captures = payment.captures ?? [];
84	      const captured = captures.length
85	        ? captures.reduce((sum, capture) => sum + toNumber(capture.amount), 0)
86	        : payment.captured_at
87	          ? toNumber(payment.amount)
88	          : 0;
89	
90	      const refunded = (payment.refunds ?? []).reduce(
91	        (sum, refund) => sum + toNumber(refund.amount),
92	        0
93	      );
94	
95	      // Round to cents: summing float amounts drifts, and the numpad would
96	      // otherwise prefill something like 29.990000000000002.
97	      const remaining = Math.max(0, Math.round((captured - refunded) * 100) / 100);
98	      if (remaining <= 0) continue;
99	
100	      refundable.push({
101	        id: payment.id,
102	        providerId: payment.provider_id,
103	        captured,
104	        refunded,
105	        refundable: remaining,
106	      });
107	    }
108	  }
109	
110	  return refundable;
111	}
